  <script>
    // ESTADO GLOBAL DE LA APLICACIÓN
    let datos = JSON.parse(localStorage.getItem('agenda_app_db')) || {
      trabajos: [],
      finanzas: []
    };

    function guardarDB() {
      localStorage.setItem('agenda_app_db', JSON.stringify(datos));
      render();
    }

    // NAVEGACIÓN ENTRE PESTAÑAS
    function cambiarPestana(pestana) {
      document.getElementById('sec-trabajos').classList.add('hidden');
      document.getElementById('sec-finanzas').classList.add('hidden');
      document.getElementById('tab-trabajos').classList.remove('active');
      document.getElementById('tab-finanzas').classList.remove('active');

      document.getElementById(`sec-${pestana}`).classList.remove('hidden');
      document.getElementById(`tab-${pestana}`).classList.add('active');
    }

    // GESTIÓN DE TRABAJOS Y TAREAS
    function guardarTrabajo(e) {
      e.preventDefault();
      const nombreInput = document.getElementById('trabajo-nombre');
      datos.trabajos.push({
        id: Date.now(),
        nombre: nombreInput.value,
        tareas: []
      });
      nombreInput.value = '';
      guardarDB();
    }

    function eliminarTrabajo(id) {
      if (confirm('¿Eliminar este trabajo y todas sus tareas asociadas?')) {
        datos.trabajos = datos.trabajos.filter(t => t.id !== id);
        guardarDB();
      }
    }

    function agregarTarea(e, trabajoId) {
      e.preventDefault();
      const form = e.target;
      const titulo = form.querySelector('.tarea-titulo').value;
      const fecha = form.querySelector('.tarea-fecha').value;

      const trabajo = datos.trabajos.find(t => t.id === trabajoId);
      if (trabajo) {
        trabajo.tareas.push({
          id: Date.now(),
          titulo,
          fecha,
          completada: false,
          notificada: false
        });
        guardarDB();
      }
    }

    function toggleTarea(trabajoId, tareaId) {
      const trabajo = datos.trabajos.find(t => t.id === trabajoId);
      if (trabajo) {
        const tarea = trabajo.tareas.find(t => t.id === tareaId);
        if (tarea) tarea.completada = !tarea.completada;
        guardarDB();
      }
    }

    function eliminarTarea(trabajoId, tareaId) {
      const trabajo = datos.trabajos.find(t => t.id === trabajoId);
      if (trabajo) {
        trabajo.tareas = trabajo.tareas.filter(t => t.id !== tareaId);
        guardarDB();
      }
    }

    // GESTIÓN FINANCIERA
    function guardarMovimiento(e) {
      e.preventDefault();
      const tipo = document.getElementById('fin-tipo').value;
      const concepto = document.getElementById('fin-concepto').value;
      const monto = parseFloat(document.getElementById('fin-monto').value);
      const fecha = document.getElementById('fin-fecha').value;

      datos.finanzas.push({
        id: Date.now(),
        tipo,
        concepto,
        monto,
        fecha
      });

      document.getElementById('fin-concepto').value = '';
      document.getElementById('fin-monto').value = '';
      guardarDB();
    }

    function eliminarMovimiento(id) {
      datos.finanzas = datos.finanzas.filter(f => f.id !== id);
      guardarDB();
    }

    // RENDERIZADO DE INTERFAZ
    function render() {
      // 1. Render Trabajos
      const contenedorTrabajos = document.getElementById('lista-trabajos');
      contenedorTrabajos.innerHTML = '';

      if (datos.trabajos.length === 0) {
        contenedorTrabajos.innerHTML = `<div class="text-center py-8 text-slate-400">No tienes proyectos o trabajos cargados. ¡Crea el primero arriba!</div>`;
      }

      datos.trabajos.forEach(trabajo => {
        const div = document.createElement('div');
        div.className = 'bg-white rounded-xl shadow-md border border-slate-200 overflow-hidden';
        div.innerHTML = `
          <div class="bg-slate-50 px-5 py-3 border-b flex justify-between items-center">
            <h3 class="font-bold text-slate-800 text-md">${trabajo.nombre}</h3>
            <button onclick="eliminarTrabajo(${trabajo.id})" class="text-rose-500 hover:text-rose-700 text-sm">
              <i class="fa-solid fa-trash"></i>
            </button>
          </div>
          <div class="p-5 space-y-4">
            <!-- Formulario Tarea -->
            <form onsubmit="agregarTarea(event, ${trabajo.id})" class="flex flex-col sm:flex-row gap-2">
              <input type="text" placeholder="Nueva tarea pendiente..." required class="tarea-titulo flex-1 p-2 border rounded-lg text-sm outline-none">
              <input type="datetime-local" required class="tarea-fecha p-2 border rounded-lg text-sm outline-none">
              <button type="submit" class="bg-slate-800 hover:bg-slate-900 text-white px-4 py-2 rounded-lg text-sm transition">
                Agregar Tarea
              </button>
            </form>
            <!-- Lista Tareas -->
            <div class="space-y-2">
              ${trabajo.tareas.length === 0 ? '<p class="text-xs text-slate-400 italic">Sin tareas pendientes.</p>' : ''}
              ${trabajo.tareas.map(tarea => `
                <div class="flex items-center justify-between p-2.5 rounded-lg border ${tarea.completada ? 'bg-slate-50 border-slate-200' : 'bg-white border-slate-300'}">
                  <div class="flex items-center space-x-3">
                    <input type="checkbox" ${tarea.completada ? 'checked' : ''} onchange="toggleTarea(${trabajo.id},${tarea.id})" class="h-4 w-4 text-blue-600 rounded">
                    <div>
                      <p class="text-sm ${tarea.completada ? 'line-through text-slate-400' : 'text-slate-800 font-medium'}">${tarea.titulo}</p>
                      <p class="text-xs text-slate-400"><i class="fa-regular fa-clock"></i> ${new Date(tarea.fecha).toLocaleString()}</p>
                    </div>
                  </div>
                  <button onclick="eliminarTarea(${trabajo.id},${tarea.id})" class="text-slate-400 hover:text-rose-500 text-xs">
                    <i class="fa-solid fa-xmark"></i>
                  </button>
                </div>
              `).join('')}
            </div>
          </div>
        `;
        contenedorTrabajos.appendChild(div);
      });

      // 2. Render Finanzas
      const tablaFinanzas = document.getElementById('tabla-finanzas');
      tablaFinanzas.innerHTML = '';

      let totalIngresos = 0;
      let totalEgresos = 0;

      datos.finanzas.forEach(f => {
        if (f.tipo === 'ingreso') totalIngresos += f.monto;
        else totalEgresos += f.monto;

        const tr = document.createElement('tr');
        tr.className = 'border-b hover:bg-slate-50';
        tr.innerHTML = `
          <td class="p-3">${f.fecha}</td>
          <td class="p-3"><span class="px-2 py-1 rounded-full text-xs font-bold ${f.tipo === 'ingreso' ? 'bg-emerald-100 text-emerald-700' : 'bg-rose-100 text-rose-700'}">${f.tipo.toUpperCase()}</span></td>
          <td class="p-3 font-medium text-slate-800">${f.concepto}</td>
          <td class="p-3 text-right font-bold ${f.tipo === 'ingreso' ? 'text-emerald-600' : 'text-rose-600'}">$${f.monto.toFixed(2)}</td>
          <td class="p-3 text-center">
            <button onclick="eliminarMovimiento(${f.id})" class="text-rose-500 hover:text-rose-700 text-xs"><i class="fa-solid fa-trash"></i></button>
          </td>
        `;
        tablaFinanzas.appendChild(tr);
      });

      document.getElementById('total-ingresos').innerText = `$${totalIngresos.toFixed(2)}`;
      document.getElementById('total-egresos').innerText = `$${totalEgresos.toFixed(2)}`;
      const balance = totalIngresos - totalEgresos;
      const balanceElem = document.getElementById('balance-neto');
      balanceElem.innerText = `$${balance.toFixed(2)}`;
      balanceElem.className = `text-2xl font-bold mt-1 ${balance >= 0 ? 'text-blue-600' : 'text-rose-600'}`;
    }

    // NOTIFICACIONES Y ALERTAS
    function solicitarPermisoNotificaciones() {
      if ('Notification' in window) {
        Notification.requestPermission().then(perm => {
          if (perm === 'granted') alert('¡Notificaciones activadas con éxito!');
        });
      } else {
        alert('Este navegador no soporta notificaciones automáticas.');
      }
    }

    function verificarAlertas() {
      const ahora = new Date().getTime();
      datos.trabajos.forEach(trabajo => {
        trabajo.tareas.forEach(tarea => {
          if (!tarea.completada && !tarea.notificada) {
            const fechaTarea = new Date(tarea.fecha).getTime();
            if (fechaTarea <= ahora) {
              if (Notification.permission === 'granted') {
                new Notification(`Recordatorio: ${tarea.titulo}`, {
                  body: `Proyecto: ${trabajo.nombre}`,
                  icon: 'https://cdn-icons-png.flaticon.com/512/3239/3239952.png'
                });
              }
              tarea.notificada = true;
              guardarDB();
            }
          }
        });
      });
    }

    setInterval(verificarAlertas, 30000); // Revisa cada 30 segundos

    // EXPORTAR E IMPORTAR COPIAS DE SEGURIDAD
    function exportarDatos() {
      const dataStr = "data:text/json;charset=utf-8," + encodeURIComponent(JSON.stringify(datos));
      const downloadAnchor = document.createElement('a');
      downloadAnchor.setAttribute("href", dataStr);
      downloadAnchor.setAttribute("download", `respaldo_agenda_${new Date().toISOString().slice(0,10)}.json`);
      document.body.appendChild(downloadAnchor);
      downloadAnchor.click();
      downloadAnchor.remove();
    }

    function importarDatos(e) {
      const file = e.target.files[0];
      if (!file) return;
      const reader = new FileReader();
      reader.onload = function(e) {
        try {
          datos = JSON.parse(e.target.result);
          guardarDB();
          alert('¡Datos importados correctamente!');
        } catch (err) {
          alert('Error al leer el archivo de respaldo.');
        }
      };
      reader.readAsText(file);
    }

    // INICIALIZACIÓN
    document.getElementById('fin-fecha').valueAsDate = new Date();
    render();
  </script>
