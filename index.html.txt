    function switchTab(tabId) {
      // Hide all contents
      document.querySelectorAll('.tab-content').forEach(el => el.classList.add('hidden'));

        el.classList.remove('bg-slate-800', 'text-cyan-400');
        el.classList.add('text-slate-400');
      });

      // Show selected tab
      document.getElementById(`tab-${tabId}`).classList.remove('hidden');
      const activeNav = document.getElementById(`nav-${tabId}`);
      activeNav.classList.add('bg-slate-800', 'text-cyan-400');

      const titles = {
        'dashboard': 'Dashboard General',
        'tasks': 'Agenda y Control de Tareas',
        'finances': 'Finanzas Personales'
      };
      document.getElementById('page-title').textContent = titles[tabId] || 'GestorPro';

    function handleSaveTask(e) {
      e.preventDefault();
      const id = document.getElementById('task-id').value;
      const title = document.getElementById('task-title').value;
      const projectId = document.getElementById('task-project').value;
      const date = document.getElementById('task-date').value;
      const time = document.getElementById('task-time').value;
      const priority = document.getElementById('task-priority').value;
      const description = document.getElementById('task-desc').value;

      if (id) {
        // Edit existing
        const task = appState.tasks.find(t => t.id === id);
        if (task) {
          task.title = title;
          task.projectId = projectId;
          task.date = date;
          task.time = time;
          task.priority = priority;
          task.description = description;
          task.alertTriggered = false; // reset alert trigger
        }
      } else {
        // Create new
        const newTask = {
          id: 'task-' + Date.now(),
          title,
          projectId,
          date,
          time,
          priority,
          description,
          completed: false,
          alertTriggered: false
        };
        appState.tasks.push(newTask);
      }

      saveDataToStorage();
      closeModal('modal-task');
      document.getElementById('form-task').reset();
      renderAll();
    }

    function toggleTaskCompletion(taskId) {
      const task = appState.tasks.find(t => t.id === taskId);
      if (task) {
        task.completed = !task.completed;
        saveDataToStorage();
        renderAll();
      }
    }

    function deleteTask(taskId) {
      if (confirm('¿Estás seguro de eliminar esta tarea?')) {
        appState.tasks = appState.tasks.filter(t => t.id !== taskId);
        saveDataToStorage();
        renderAll();
      }
    }

    function renderTasks() {
      const container = document.getElementById('tasks-container');
      const selectedProject = document.getElementById('filter-project').value;

      let filtered = appState.tasks.filter(task => {
        const matchesStatus = activeFilterStatus === 'ALL' ? true :
                              activeFilterStatus === 'COMPLETED' ? task.completed : !task.completed;
        const matchesProject = selectedProject === 'ALL' ? true : task.projectId === selectedProject;
        return matchesStatus && matchesProject;
      });

      // Sort by date & time
      filtered.sort((a, b) => new Date(`${a.date}T${a.time}`) - new Date(`${b.date}T${b.time}`));

      document.getElementById('task-count-badge').textContent = `${filtered.length} tarea(s)`;

      if (filtered.length === 0) {
        container.innerHTML = `
          <div class="p-8 text-center text-slate-500">
            <i class="fa-regular fa-folder-open text-3xl mb-2 block"></i>
            <span>No se encontraron tareas registradas o pendientes.</span>
          </div>`;
        return;
      }

      const priorityBadges = {
        HIGH: '<span class="px-2 py-0.5 text-[10px] rounded-full bg-rose-500/10 text-rose-400 border border-rose-500/20 font-semibold">Alta</span>',
        MEDIUM: '<span class="px-2 py-0.5 text-[10px] rounded-full bg-amber-500/10 text-amber-400 border border-amber-500/20 font-semibold">Media</span>',
        LOW: '<span class="px-2 py-0.5 text-[10px] rounded-full bg-slate-500/10 text-slate-400 border border-slate-500/20 font-semibold">Baja</span>'
      };

      container.innerHTML = filtered.map(task => {
        const project = appState.projects.find(p => p.id === task.projectId);
        const projectBadge = project ? 
          `<span class="px-2 py-0.5 text-[10px] rounded-md text-white font-medium" style="background-color: ${project.color}">${project.name}</span>` :

    function renderTransactions() {
      const tbody = document.getElementById('transactions-table-body');
      const filter = document.getElementById('filter-tx-type').value;

      let filtered = appState.transactions.filter(t => filter === 'ALL' ? true : t.type === filter);
      filtered.sort((a, b) => new Date(b.date) - new Date(a.date));

      if (filtered.length === 0) {
        tbody.innerHTML = `<tr><td colspan="6" class="p-6 text-center text-slate-500">No hay movimientos registrados.</td></tr>`;
        return;
      }

      tbody.innerHTML = filtered.map(tx => {
        const project = appState.projects.find(p => p.id === tx.projectId);
        const isIncome = tx.type === 'INCOME';
        
        return `
          <tr class="hover:bg-slate-800/30 transition">
            <td class="p-3.5 text-xs text-slate-400">${tx.date}</td>
            <td class="p-3.5 font-medium text-slate-200">${tx.description}</td>
            <td class="p-3.5"><span class="px-2 py-0.5 text-xs rounded-md bg-slate-800 text-slate-300 border border-slate-700">${tx.category}</span></td>
            <td class="p-3.5 text-xs text-slate-400">${project ? project.name : '-'}</td>
            <td class="p-3.5 text-right font-semibold ${isIncome ? 'text-emerald-400' : 'text-rose-400'}">
              ${isIncome ? '+' : '-'}$${tx.amount.toFixed(2)}
            </td>
            <td class="p-3.5 text-center">
              <button onclick="deleteTransaction('${tx.id}')" class="text-slate-500 hover:text-rose-400 transition">
                <i class="fa-solid fa-trash-can"></i>
              </button>
            </td>
          </tr>
        `;
      }).join('');
    }

    // DASHBOARD & CHART RENDERING
    function renderDashboardMetrics() {
      const pendingTasks = appState.tasks.filter(t => !t.completed).length;
      const totalIncome = appState.transactions.filter(t => t.type === 'INCOME').reduce((acc, t) => acc + t.amount, 0);
      const totalExpenses = appState.transactions.filter(t => t.type === 'EXPENSE').reduce((acc, t) => acc + t.amount, 0);
      const netBalance = totalIncome - totalExpenses;

      document.getElementById('dash-pending-count').textContent = pendingTasks;
      document.getElementById('dash-projects-count').textContent = appState.projects.length;
      document.getElementById('dash-income-total').textContent = `$${totalIncome.toFixed(2)}`;
      document.getElementById('dash-net-balance').textContent = `$${netBalance.toFixed(2)}`;
      
      const balanceEl = document.getElementById('dash-net-balance');
      balanceEl.className = `text-3xl font-extrabold ${netBalance >= 0 ? 'text-emerald-400' : 'text-rose-400'}`;

      // Update Financial Tab cards
      document.getElementById('fin-total-income').textContent = `$${totalIncome.toFixed(2)}`;
      document.getElementById('fin-total-expenses').textContent = `$${totalExpenses.toFixed(2)}`;
      document.getElementById('fin-total-balance').textContent = `$${netBalance.toFixed(2)}`;

      // Render upcoming tasks on Dashboard
      renderUpcomingTasks();
    }

    function renderUpcomingTasks() {
      const container = document.getElementById('upcoming-tasks-list');
      const upcoming = appState.tasks
        .filter(t => !t.completed)
        .sort((a, b) => new Date(`${a.date}T${a.time}`) - new Date(`${b.date}T${b.time}`))
        .slice(0, 4);

      if (upcoming.length === 0) {
        container.innerHTML = `<p class="text-slate-500 text-xs text-center py-4">No hay tareas pendientes.</p>`;
        return;
      }

      container.innerHTML = upcoming.map(t => `
        <div class="p-3 rounded-xl bg-slate-800/50 border border-slate-700/50 flex items-center justify-between">
          <div>
            <h5 class="text-xs font-semibold text-slate-200">${t.title}</h5>
            <span class="text-[10px] text-slate-400"><i class="fa-regular fa-clock mr-1"></i>${t.date} ${t.time} hs</span>

          <button onclick="toggleTaskCompletion('${t.id}')" class="text-xs px-2 py-1 bg-slate-700 hover:bg-cyan-600 text-slate-200 rounded-lg transition">

    function renderChart() {
      const ctx = document.getElementById('financeChart').getContext('2d');
      const totalIncome = appState.transactions.filter(t => t.type === 'INCOME').reduce((acc, t) => acc + t.amount, 0);
      const totalExpenses = appState.transactions.filter(t => t.type === 'EXPENSE').reduce((acc, t) => acc + t.amount, 0);

      if (financeChartInstance) {
        financeChartInstance.destroy();
      }

      financeChartInstance = new Chart(ctx, {
        type: 'bar',
        data: {
          labels: ['Ingresos', 'Gastos'],
          datasets: [{
            label: 'Monto Total ($)',
            data: [totalIncome, totalExpenses],
            backgroundColor: ['rgba(16, 185, 129, 0.6)', 'rgba(244, 63, 94, 0.6)'],
            borderColor: ['#10b981', '#f43f5e'],
            borderWidth: 2,
            borderRadius: 12
          }]
        },
        options: {
          responsive: true,
          maintainAspectRatio: false,
          plugins: {
            legend: { display: false }
          },
          scales: {
            y: {
              beginAtZero: true,
              grid: { color: 'rgba(51, 65, 85, 0.3)' },
              ticks: { color: '#94a3b8' }
            },
            x: {
              grid: { display: false },
              ticks: { color: '#94a3b8' }
            }
          }
        }
      });
    }

    // WEB NOTIFICATION & REALTIME ALERT SYSTEM
    function checkNotificationPermissions() {
      if (!("Notification" in window)) return;
      
      const badge = document.getElementById('notif-badge');
      const btn = document.getElementById('btn-request-notif');

      if (Notification.permission === "granted") {
        badge.textContent = "Activo";
        badge.className = "px-2 py-0.5 rounded-full text-[10px] bg-emerald-500/20 text-emerald-300 border border-emerald-500/30";
        btn.classList.add('hidden');
      } else if (Notification.permission === "denied") {
        badge.textContent = "Bloqueado";
        badge.className = "px-2 py-0.5 rounded-full text-[10px] bg-rose-500/20 text-rose-300 border border-rose-500/30";
      }
    }

    function requestNotificationPermission() {
      if (!("Notification" in window)) {
        alert("Este navegador no soporta notificaciones de escritorio.");
        return;
      }
      Notification.requestPermission().then(permission => {
        checkNotificationPermissions();
      });
    }

    function checkScheduledAlerts() {
      const now = new Date();
      
      appState.tasks.forEach(task => {
        if (task.completed || task.alertTriggered) return;

        const taskDateTime = new Date(`${task.date}T${task.time}`);
        // Trigger alert if current time is equal or past the task target time within a 5-minute window
        const diffMs = taskDateTime - now;

        if (diffMs <= 0 && Math.abs(diffMs) < 300000) { // triggered within 5 mins
          task.alertTriggered = true;
          saveDataToStorage();

          // Trigger Desktop Notification
          if ("Notification" in window && Notification.permission === "granted") {
            new Notification(`⏰ RECORDATORIO: ${task.title}`, {
              body: `Fecha: ${task.date} - ${task.time} hs\nPrioridad: ${task.priority}`,
              icon: 'https://cdn-icons-png.flaticon.com/512/3602/3602145.png'
            });
          } else {
            alert(`⏰ RECORDATORIO DE TAREA:\n\n${task.title}\nHora: ${task.time}`);
          }
